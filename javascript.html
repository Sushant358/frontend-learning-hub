2. Variables and naming

Use `const` by default and `let` when reassignment is needed. Avoid `var` in modern code unless you specifically need its older function-scoping behavior.

const firstName = "Asha";
let score = 0;
score = score + 10;

// Naming: camelCase is common for variables and functions
const totalMarks = 450;

3. Data types and type conversion

Primitive types include string, number, bigint, boolean, undefined, null, and symbol. Objects, arrays, and functions are reference types.

const name = "Ravi";       // string
const age = 20;             // number
const isStudent = true;     // boolean
let result;                 // undefined
const emptyValue = null;    // null
const skills = ["HTML", "CSS"]; // array
const student = { name, age };  // object

console.log(Number("42")); // 42
console.log(String(42));   // "42"
console.log(Boolean(1));   // true

4. Operators and comparisons

Learn arithmetic, assignment, comparison, logical, nullish coalescing, optional chaining, and ternary operators. Prefer strict equality (`===`) in most cases.

const total = 10 + 5;
const remainder = 10 % 3;
const power = 2 ** 3;

let count = 0;
count += 1;

console.log(5 === "5"); // false
console.log(true && false);
const label = score >= 40 ? "Pass" : "Try again";
const userName = user?.profile?.name ?? "Guest";

5. Conditions

Use `if/else` for branching and `switch` when comparing one value against several cases.

const marks = 78;
if (marks >= 90) {
  console.log("Excellent");
} else if (marks >= 40) {
  console.log("Pass");
} else {
  console.log("Try again");
}

switch (day) {
  case "Monday": console.log("Start"); break;
  default: console.log("Another day");
}

6. Loops

Loops repeat work. Use `for` when the count is known, `while` when a condition controls repetition, and `for...of` to iterate iterable values.

for (let i = 1; i <= 5; i++) {
  console.log(i);
}

let n = 1;
while (n <= 3) {
  console.log(n);
  n++;
}

for (const skill of ["HTML", "CSS", "JS"]) {
  console.log(skill);
}

7. Functions

Functions package reusable logic. Understand declarations, expressions, arrow functions, parameters, return values, default parameters, and callbacks.

function add(a, b = 0) {
  return a + b;
}

const multiply = (a, b) => a * b;

function greet(name = "Guest") {
  return `Hello, ${name}!`;
}

console.log(add(4, 6));

8. Scope and hoisting

Scope determines where a variable can be accessed. Learn global, function, and block scope, lexical scope, closures, and the different behavior of `let`, `const`, and `var`.

const globalMessage = "Hello";

function showMessage() {
  const localMessage = "Inside function";
  if (true) {
    let blockValue = 10;
    console.log(blockValue);
  }
  console.log(localMessage);
}

// A closure remembers its outer lexical environment
function makeCounter() {
  let count = 0;
  return () => ++count;
}
const nextCount = makeCounter();

9. Arrays and array methods

Arrays store ordered collections. Practice indexing, `length`, `at()`, adding/removing elements, and methods such as `map`, `filter`, `find`, `some`, and `every`.

const marks = [70, 85, 92];
marks.push(96);
console.log(marks[0], marks.at(-1));

const doubled = marks.map(mark => mark * 2);
const passed = marks.filter(mark => mark >= 40);
const firstHigh = marks.find(mark => mark > 90);

10. Objects and destructuring

Objects store data as key-value pairs. Learn properties, methods, `this`, optional chaining, object destructuring, spread syntax, and useful `Object` methods.

const student = {
  name: "Asha",
  marks: 88,
  greet() { return `Hi, I'm ${this.name}`; }
};

const { name: studentName, marks = 0 } = student;
const updatedStudent = { ...student, marks: 95 };
console.log(Object.keys(updatedStudent));

11. Map, filter, and reduce

These array methods are essential for transforming, selecting, and combining data. They return new results rather than requiring manual index management.

const prices = [100, 200, 300];
const discounted = prices.map(price => price * 0.9);
const expensive = prices.filter(price => price >= 200);
const total = prices.reduce((sum, price) => sum + price, 0);

12. DOM selection and manipulation

The DOM is the browser's object representation of a document. Select elements, update text or classes, create nodes, and respond to events.

const heading = document.querySelector("#title");
heading.textContent = "Welcome!";
heading.classList.add("highlight");

const item = document.createElement("li");
item.textContent = "Learn JavaScript";
document.querySelector("#list").append(item);

13. Events and forms

Events let your page react to clicks, typing, submission, and more. Prefer `addEventListener()` and use event delegation for lists of dynamic elements.

const button = document.querySelector("#change-btn");
button.addEventListener("click", () => {
  document.body.classList.toggle("dark");
});

const form = document.querySelector("#signup-form");
form.addEventListener("submit", event => {
  event.preventDefault();
  console.log("Form handled by JavaScript");
});

14. Asynchronous JavaScript

JavaScript uses the event loop to coordinate synchronous work and asynchronous tasks. Learn callbacks, promises, `async/await`, timers, and error handling.

async function loadUsers() {
  try {
    const response = await fetch("https://jsonplaceholder.typicode.com/users");
    if (!response.ok) throw new Error("Request failed");
    const users = await response.json();
    console.log(users);
  } catch (error) {
    console.error(error.message);
  }
}
loadUsers();

16. Browser storage and APIs

Browser APIs provide capabilities such as local storage, session storage, timers, URL handling, and clipboard access. Do not store secrets in browser storage.

localStorage.setItem("theme", "dark");
const savedTheme = localStorage.getItem("theme");

sessionStorage.setItem("step", "2");

// JSON is useful for storing structured data
const settings = { theme: "dark", fontSize: 16 };
localStorage.setItem("settings", JSON.stringify(settings));
const restored = JSON.parse(localStorage.getItem("settings") || "{}");

18. Classes and object-oriented JavaScript

Classes provide syntax for creating objects with shared methods. Also understand prototypes, inheritance, and composition.

class Person {
  constructor(name) {
    this.name = name;
  }
  introduce() {
    return `I'm ${this.name}`;
  }
}

class Student extends Person {
  study() {
    return `${this.name} is studying`;
  }
}

const learner = new Student("Asha");

19. Practice project: interactive task list

Build a task list that adds tasks, marks them complete, and removes them. Practice DOM creation, events, arrays, and local storage.

const tasks = [];
function addTask(text) {
  const cleanText = text.trim();
  if (!cleanText) return;
  tasks.push({ id: Date.now(), text: cleanText, done: false });
  renderTasks();
}

function renderTasks() {
  const list = document.querySelector("#tasks");
  list.replaceChildren();
  for (const task of tasks) {
    const li = document.createElement("li");
    li.textContent = task.text;
    list.append(li);
  }
}
